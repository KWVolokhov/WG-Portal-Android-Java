Модель отработала, обновила код — скрипт сам делает коммит с понятным описанием (например, «Ollama: добавлено логирование»).Вы сможете запускать генерацию циклично, а потом просто просматривать историю коммитов в Git.Вот как будет выглядеть обновленный скрипт с поддержкой Git (используется встроенный модуль subprocess):pythonimport re
import subprocess
import ollama

# НАСТРОЙКИ
TARGET_FILE = "target_script.py"
MODEL_NAME = "qwen2.5-coder:7b"
USER_PROMPT = "Оптимизируй циклы и добавь обработку исключений."

# 1. Читаем исходный код
with open(TARGET_FILE, "r", encoding="utf-8") as f:
    source_code = f.read()

system_instruction = (
    "Ты — профессиональный Python разработчик. Исправь или дополни код по запросу пользователя. "
    "Возвращай ТОЛЬКО готовый рабочий код внутри блока ```python ... ```. "
    "Не пиши никаких лишних приветствий или объяснений."
)

print("🤖 Отправка запроса в Ollama (генерация на CPU)...")

# 2. Запрос к Ollama
response = ollama.generate(
    model=MODEL_NAME,
    system=system_instruction,
    prompt=f"Исходный код:\n\n{source_code}\n\nЗадача: {USER_PROMPT}",
    options={"num_ctx": 8192, "temperature": 0.2}
)

ai_text = response['response']
code_match = re.search(r"```python\s*(.*?)\s*```", ai_text, re.DOTALL)

if code_match:
    cleaned_code = code_match.group(1)
    
    # 3. Перезаписываем файл
    with open(TARGET_FILE, "w", encoding="utf-8") as f:
        f.write(cleaned_code)
    print(f"✅ Файл '{TARGET_FILE}' обновлен решением от Ollama.")
    
    # 4. Автоматический коммит в Git
    try:
        subprocess.run(["git", "add", TARGET_FILE], check=True)
        commit_message = f"Ollama refactor: {USER_PROMPT}"
        subprocess.run(["git", "commit", "-m", commit_message], check=True)
        print(f"📦 Изменения автоматически закоммичены в ветку с сообщением: '{commit_message}'")
    except subprocess.CalledProcessError as e:
        print(f"⚠️ Git коммит не удался (возможно, нет изменений или вы вне репозитория): {e}")
else:
    print("❌ Ошибка: Модель не вернула код в блоке ```python ... ```.")

Это действительно удобнее: скрипт сам создаст чистое пространство для экспериментов нейросети, а вы уже сами решите, делать git commit или git restore.Для этого мы используем встроенную библиотеку subprocess. Скрипт будет брать ваш запрос (промпт), делать из него короткое безопасное имя для ветки (например, ollama-add-logging) и переключаться на неё до того, как затрёт файл.Вот готовое решение:pythonimport re
import subprocess
import ollama

# НАСТРОЙКИ
TARGET_FILE = "target_script.py"
MODEL_NAME = "qwen2.5-coder:7b"
USER_PROMPT = "Добавь валидацию входящих данных и обработку ошибок"


def create_git_branch(prompt_text):
    """Создает безопасное имя ветки из промпта и переключается на нее"""
    # Переводим в нижний регистр, заменяем пробелы на дефисы и убираем спецсимволы
    clean_name = re.sub(r'[^a-zA-Z0-9\s-]', '', prompt_text.lower())
    # Берем первые несколько слов для читаемости
    slug = "-".join(clean_name.split()[:4])
    branch_name = f"ollama-{slug}"
    
    try:
        # Создаем и переключаемся на новую ветку (-b)
        subprocess.run(["git", "checkout", "-b", branch_name], check=True, capture_output=True, text=True)
        print(f"🌿 Создана и активирована новая ветка: {branch_name}")
    except subprocess.CalledProcessError:
        # Если ветка уже существует, просто переключаемся на нее
        subprocess.run(["git", "checkout", branch_name], check=True, capture_output=True, text=True)
        print(f"🔄 Переключено на существующую ветку: {branch_name}")


# 1. Сначала подготавливаем Git-окружение
create_git_branch(USER_PROMPT)

# 2. Читаем исходный код
with open(TARGET_FILE, "r", encoding="utf-8") as f:
    source_code = f.read()

system_instruction = (
    "Ты — профессиональный Python разработчик. Исправь или дополни код по запросу пользователя. "
    "Возвращай ТОЛЬКО готовый рабочий код внутри блока ```python ... ```. "
    "Не пиши никаких лишних приветствий или объяснений."
)

print("🤖 Отправка запроса в Ollama (генерация на CPU, нужно подождать)...")

# 3. Запрос к Ollama
response = ollama.generate(
    model=MODEL_NAME,
    system=system_instruction,
    prompt=f"Исходный код:\n\n{source_code}\n\nЗадача: {USER_PROMPT}",
    options={"num_ctx": 8192, "temperature": 0.2}
)

ai_text = response['response']
code_match = re.search(r"```python\s*(.*?)\s*```", ai_text, re.DOTALL)

if code_match:
    cleaned_code = code_match.group(1)
    
    # 4. Перезаписываем файл в новой ветке
    with open(TARGET_FILE, "w", encoding="utf-8") as f:
        f.write(cleaned_code)
    print(f"✅ Файл '{TARGET_FILE}' успешно изменен моделью.")
    print("👉 Теперь вы можете проверить `git diff` и сделать коммит вручную.")
else:
    print("❌ Ошибка: Модель не вернула код в блоке ```python ... ```.")
Используйте код с осторожностью.Как это работает в процессе:Вы пишете промпт, например: "Добавь валидацию входящих данных".Скрипт мгновенно делает git checkout -b ollama-dobav-validaciyu (если промпт на русском, он уберет спецсимволы, поэтому для идеальных имен веток промпт в переменную USER_PROMPT лучше на время теста вписать на английском, либо скрипт назовет её просто ollama-).Модель генерирует код и меняет файл.Вы открываете терминал, пишете git diff, смотрите, что i5-процессор вам там нагенерировал, и если все ок — коммитите.Если планируете писать промпты на русском языке, нужно ли добавить в функцию create_git_branch транслитерацию (чтобы ветки назывались ollama-dobav-validaciyu, а не просто ollama-)?
